* Prelude
#+begin_src js
import * as b from './borth-core.js'

const { find, currentScope, constant, nextToken, define, docol } = b
const effect = (name, fn) => b.word(c => (fn(c), c), { type: 'effect', name })
const def = (name, fn) => b.def(fn, { name })
const syn = (name, fn) => b.syn(fn, { name })
const staging = ctx => currentScope(ctx).staging
const resolveNs = (ctx, name) => {
  if (!(name in ctx.vocabs)) {
    ctx.vocabs[name] = new Map()
  }
  return ctx.vocabs[name]
}
const enter = (ctx, namespace) => {
  if (!(namespace in ctx.vocabs)) {
    ctx.vocabs[namespace] = new Map()
  }
  ctx.ns = ctx.vocabs[namespace]
}
const search = (ctx, names) => {
  ctx.search = names.map(name => resolveNs(ctx, name))
}
const install = (ctx, ...pkgs) => {
  const old = ctx.ns
  for (const [name, ...words] of pkgs) {
    const v = resolveNs(ctx, name)
    ctx.ns = v
    for (const w of words) b.define(ctx, w)
    ctx.ns = old
  }
}
export const prelude = [
  'prelude',
  def('dup', a => [a, a]),
  def('drop', _ => {}),
  def('swap', (a, b) => [b, a]),
  def('rot', (a, b, c) => [b, c, a]),
  def('over', (a, b) => [a, b, a]),
  def('+', (a, b) => [a + b]),
  def('-', (a, b) => [a - b]),
  def('*', (a, b) => [a * b]),
  def('/', (a, b) => [a / b]),
  def('mod', (a, b) => [a % b]),
  def('=', (a, b) => [a === b]),
  def('<', (a, b) => [a < b]),
  def('>', (a, b) => [a > b]),
  def('<=', (a, b) => [a <= b]),
  def('>=', (a, b) => [a >= b]),
  def('true', () => true),
  def('false', () => false),
  def('nil', () => null),
  def('@', word => [b.props(word).slot ?? null]),
  def('!', (value, word) => [b.props(word, { slot: value })]),
  effect('.', c => c.emit(b.pop(c, 1))),
  syn('in:', c => {
    enter(c, nextToken(c))
    return c
  }),
  syn('using:', c => {
    const names = b.parseUntil(c, ';').split(/\s+/)
    search(c, names)
    return c
  }),
  syn(':', c => {
    const name = nextToken(c)
    b.enterCompilationScope(c, async (ctx, { staging }) => {
      await define(ctx, await docol(name, staging))
    })
    return c
  }),
  syn(';', async c => {
    await b.exitCompilationScope(c)
    return c
  }),
]

* Syntax
#+begin_src js
export const syntax = [
  'syntax',
  syn('str:', c => staging(c).push(constant(nextToken(c)))),
  syn('\\', c => staging(c).push(constant(find(c, nextToken(c))))),
  syn('macro:', async c => {
    const name = nextToken(c)
    b.enterCompilationScope(c, async (ctx, {staging}) => {
      const w = b.word(
        async (c) => {
          const old = c.stack;
          c.stack = [];
          await b.evaluate(c, staging)
          const toCompile = c.stack.map(constant);
          b.stage(ctx, ...toCompile)
          c.stack = old;
          return c
        },
        {type: 'macro', parsing: true, name}
      );
      await define(ctx, w);
    })
    return c
  })
]
